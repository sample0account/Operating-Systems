#include <stdio.h>
#include <stdlib.h>
#define MAX 20

int list[MAX], n, cdr;

void display()
{
    int i;
    printf("\nProcesses -->");
    for (i = 1; i <= n; i++)
        printf("\t%d", i);
    printf("\nAlive -->");
    for (i = 1; i <= n; i++)
        printf("\t%d", list[i]);
    printf("\nCoordinator is: %d", cdr);
}

void ring()
{
    int msg[20], k, i;
    int ch, crash, activate, gid, flag, subcdr;

    do
    {
        printf("\n1. Crash\n2. Activate\n3. Display\n4. Exit\nEnter your choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
        case 1:
            printf("\nEnter process no. to crash: ");
            scanf("%d", &crash);

            if (list[crash])
                list[crash] = 0;
            else
            {
                printf("\nProcess is already dead!");
                break;
            }

            do
            {
                printf("\nEnter election generator ID: ");
                scanf("%d", &gid);
                if (gid == cdr || list[gid] == 0)
                {
                    printf("\nEnter a valid generator ID: ");
                }
            } while (gid == cdr || list[gid] == 0);

            k = 1;

            if (crash == cdr)
            {
                msg[k++] = gid;

                for (i = (gid + 1) % n; i != gid; i = (i + 1) % n)
                {
                    if (list[i])
                    {
                        printf("\nMessage is sent to %d, k = %d", i, k);
                        msg[k++] = i;
                    }
                }

                subcdr = 0;
                for (i = 1; i < k; i++)
                {
                    printf("msg::%d\n", msg[i]);
                    if (subcdr < msg[i])
                    {
                        subcdr = msg[i];
                    }
                }
                cdr = subcdr;
            }
            display();
            break;

        case 2:
            printf("\nEnter process no. to activate: ");
            scanf("%d", &activate);

            if (!list[activate])
                list[activate] = 1;
            else
            {
                printf("\nProcess is already alive!");
                break;
            }

            if (activate == n)
            {
                cdr = n;
                break;
            }

            flag = 0;

            // Start election in a circular manner
            for (i = (activate + 1) % n; i != activate; i = (i + 1) % n)
            {
                printf("\nMessage is sent from %d to %d", activate, i);
                if (list[i])
                {
                    printf("\nResponse is sent from %d to %d", i, activate);
                    subcdr = i; // Update sub-coordinator
                    flag = 1;
                }
            }

            if (!flag)
            {
                // If no higher process is found, the activated process becomes coordinator
                cdr = activate;
            }
            else
            {
                // Otherwise, the highest active process becomes the coordinator
                cdr = subcdr;
            }
            display();
            break;

        case 3:
            display();
            break;

        case 4:
            break;
        }
    } while (ch != 4);
}

int main()
{
    int i;
    printf("\nEnter number of processes: ");
    scanf("%d", &n);

    for (i = 1; i <= n; i++)
    {
        printf("\nEnter if process %d is Alive or not (0/1): ", i);
        scanf("%d", &list[i]);
        if (list[i])
        {
            cdr = i;
        }
    }

    display();
    ring();
    return 0;
}


### **Step-by-Step Algorithm for the Ring Election Algorithm**  

1. **Define global variables** – Store process states in `list[MAX]`, number of processes in `n`, and coordinator in `cdr`.  
2. **Implement `display` function** – Print the list of processes, their status (alive/dead), and the current coordinator.  
3. **Implement `ring` function** – Provide an interactive menu to crash, activate processes, display status, or exit.  
4. **Crash a process** – Read the process ID to crash and mark it as dead if it's currently alive.  
5. **Start election if the coordinator crashes** – Ask for an election initiator process (`gid`).  
6. **Validate election initiator** – Ensure the initiator is alive and not the coordinator.  
7. **Initialize message array** – Store processes participating in the election.  
8. **Send messages in a ring structure** – The initiator sends messages to the next alive process in a circular order.  
9. **Collect responses from active processes** – Track the highest process ID in the ring.  
10. **Determine the new coordinator** – The process with the highest ID becomes the new coordinator.  
11. **Display updated state after election** – Show the new list of processes and the elected coordinator.  
12. **Activate a process** – Read the process ID to activate and mark it as alive if it was dead.  
13. **Start an election if a process is activated** – The activated process starts a circular election.  
14. **Send messages to higher processes** – The activated process contacts active processes in a circular order.  
15. **Check for responses** – If an active process responds, it remains the coordinator; otherwise, the activated process becomes the coordinator.  
16. **Display updated state after activation** – Show the new process list and coordinator.  
17. **Display option** – Allow users to view the current system state at any time.  
18. **Exit option** – Provide an option to terminate the program.  
19. **Implement `main` function** – Read the number of processes from the user.  
20. **Initialize process states** – Ask if each process is alive or dead and set the initial coordinator as the highest alive process.  
21. **Call `display` and `ring` functions** – Show the initial state and start the interactive election handling.
